			add_timed_idea = {
				idea = ARM_war_for_independence
				days = 120
			}

				idea = ARM_militia_army
				days = 120
			}

				days = 120
			}

			}

			target = ARM
			type = annex_everything
		}
		hidden_effect = {
			ARM = {
				custom_override_tooltip = {
					tooltip = {
						localization_key = tooltip_create_division_template
						NAME = "Volontaires Internationaux"
						LINE_1 = { localization_key = line_battalion_tt NAME = militia ICON = GFX_unit_militia_icon_small COUNT = 9 }
						SUPPORT_1 = { localization_key = support_battalion_tt NAME = assault_battalion ICON = GFX_unit_blackshirt_assault_battalion_icon_small }
						SUPPORT_2 = { localization_key = support_battalion_tt NAME = artillery ICON = GFX_unit_artillery_icon_small }
					}
					if = {
						limit = { has_template = "Volontaires Internationaux" }
						delete_unit_template_and_units = { division_template = "Volontaires Internationaux" disband = yes } #no cheating
					}
					division_template = {
						name = "Volontaires Internationaux"
						template_counter = 68
						role = infantry
						regiments = {
							militia = { x = 0 y = 0 }
							militia = { x = 0 y = 1 }

							militia = { x = 1 y = 2 }
							militia = { x = 2 y = 0 }
							militia = { x = 2 y = 1 }

						}
						support = {
							blackshirt_assault_battalion = { x = 0 y = 0 }
							artillery = { x = 0 y = 1 }
						}
					}
				}
				custom_effect_tooltip = generic_skip_one_line_tt

					limit = { is_capital = yes }

						tooltip = {
							localization_key = tooltip_spawn_units
							COUNT = 18
							TYPE = "Volontaires Internationaux"
