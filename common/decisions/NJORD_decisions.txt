	CHI_lower_debt_decision = {
        allowed = {
            OR = {
				has_idea = CHI_national_debt_1

				has_idea = CHI_national_debt_3
				has_idea = CHI_national_debt_4
				has_idea = CHI_national_debt_5
			}
        }
		visible = {
            OR = {
				has_idea = CHI_national_debt_1
				has_idea = CHI_national_debt_2
				has_idea = CHI_national_debt_3
				has_idea = CHI_national_debt_4
				has_idea = CHI_national_debt_5
			}
        }
        icon = GFX_decision_gre_paying_ifc_debt
        days_remove = 100
        modifier = {
            consumer_goods_factor = 0.1
			political_power_gain = -0.1
        }
        remove_effect = {
			if = {
				limit = { 
					has_idea = CHI_national_debt_1
				}
				swap_ideas = {
					remove_idea = CHI_national_debt_1
					add_idea = CHI_national_debt_2
				}
			}
			else_if = {
				limit = { 
					has_idea = CHI_national_debt_2
				}
				swap_ideas = {
					remove_idea = CHI_national_debt_2
					add_idea = CHI_national_debt_3
				}
			}
			else_if = {
				limit = { 
					has_idea = CHI_national_debt_3
				}
				swap_ideas = {
					remove_idea = CHI_national_debt_3
					add_idea = CHI_national_debt_4
				}
			}
			else_if = {
				limit = { 
					has_idea = CHI_national_debt_4
				}
				swap_ideas = {
					remove_idea = CHI_national_debt_4
					add_idea = CHI_national_debt_5
				}
			}
			else_if = {
				limit = { 
					has_idea = CHI_national_debt_5
				}
				remove_ideas = CHI_national_debt_5
			}
        }
    }
